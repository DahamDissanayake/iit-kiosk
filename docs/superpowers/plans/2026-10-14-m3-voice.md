# M3: Voice Input Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A spoken English question on the laptop mic (and on the Pi's speakerphone) gets a spoken answer, with live transcript in the bubble, the mic muted while the bot talks, and the touch menu taking over when the mic fails. Pi benchmarks decide whether voice stays in kiosk-core or runs as kiosk-voice.

**Architecture:** `Microphone` (sounddevice) delivers 32 ms frames to a worker thread. Silero VAD (ONNX) scores each frame; a pure `Segmenter` turns scores into speech start/end. Moonshine (ONNX) transcribes each finished segment, plus a growing partial while the visitor speaks. `VoicePipeline` posts `speech_started`, `partial`, `final` and `mic_status` to a `VoiceSink`: the runtime itself (in-process mode) or a ZeroMQ publisher (service mode). Piper TTS stays in kiosk-core in both modes.

**Tech Stack:** sounddevice, numpy, onnxruntime (Silero VAD v5), Moonshine ONNX, pyzmq.

**Spec:** `docs/superpowers/specs/2026-10-14-mvp-build-design.md`; baseline sections 4.3, 6, 6.1, 7.2, 11. Builds on the M1 and M2 plans.

## Changes to the baseline

- **TTS stays in kiosk-core.** Baseline 6.1 puts TTS in kiosk-voice. Keeping Piper next to the controller avoids shipping audio over ZeroMQ twice; kiosk-voice, when used, does capture, VAD and STT only. The `tts.*` topics are not used.
- **STT model.** The baseline names Moonshine Voice Small Streaming. Task 5 implements the Moonshine ONNX `base` model, whose API is stable, behind a `Transcriber` interface, and then evaluates Small Streaming. Small Streaming replaces it if it is faster on the Pi.

## Review Focus

1. **The bot's own voice.** Frames captured while the mic is muted must never reach VAD or STT, including frames already queued when the mute arrives. (Task 6: `test_frames_queued_before_mute_are_dropped`)
2. **Short noises.** A cough or a door bang (under ~100 ms of speech) must not start a turn. (Task 4: `test_short_noise_does_not_start_speech`)
3. **Visitor speaking past 15 s.** The segment is cut at the cap and transcribed, not held forever. (Task 4: `test_long_speech_is_cut_at_the_cap`)
4. **Speakerphone unplugged mid-conversation.** The touch menu appears within about 2 s, and voice comes back on its own when the device returns. (Task 6: `test_mic_stall_reports_failure_and_recovery`)
5. **Empty transcript.** VAD fired but STT heard nothing: the silence timer resumes, so the visitor still gets the one follow-up. (Task 2: `test_empty_transcript_resumes_silence_timer`)

```python
"""Local ZeroMQ links between kiosk services (baseline 7.2): [topic, JSON] frames."""

import json

import zmq
import zmq.asyncio


class Publisher:
    def __init__(self, ctx: zmq.asyncio.Context, endpoint: str) -> None:
        self.sock = ctx.socket(zmq.PUB)
        self.sock.setsockopt(zmq.SNDHWM, 100)
        self.sock.setsockopt(zmq.LINGER, 0)
        self.sock.bind(endpoint)

    async def publish(self, topic: str, payload: dict) -> None:
        await self.sock.send_multipart([topic.encode(), json.dumps(payload).encode()])

    def close(self) -> None:
        self.sock.close()


class Subscriber:
    """ZeroMQ filters by prefix, so recv() also checks the topic matches exactly."""

    def __init__(self, ctx: zmq.asyncio.Context, endpoints: list[str], topics: list[str]) -> None:
        self.topics = set(topics)
        self.sock = ctx.socket(zmq.SUB)
        self.sock.setsockopt(zmq.RCVHWM, 100)
        self.sock.setsockopt(zmq.LINGER, 0)
        for endpoint in endpoints:
            self.sock.connect(endpoint)
        for topic in topics:
            self.sock.setsockopt(zmq.SUBSCRIBE, topic.encode())

    async def recv(self) -> tuple[str, dict]:
        while True:
            raw_topic, raw = await self.sock.recv_multipart()
            topic = raw_topic.decode()
            if topic in self.topics:
                return topic, json.loads(raw)

    def close(self) -> None:
        self.sock.close()
```

```python
"""16 kHz mono WAV helpers for tests, benchmarks and replay. Never used on visitor audio."""

import wave
from pathlib import Path

import numpy as np

RATE = 16000


def read_wav(path: Path) -> np.ndarray:
    with wave.open(str(path), "rb") as w:
        if w.getframerate() != RATE:
            raise ValueError(f"{path}: need {RATE} Hz, got {w.getframerate()}")
        if w.getsampwidth() != 2:
            raise ValueError(f"{path}: need 16-bit PCM")
        raw = np.frombuffer(w.readframes(w.getnframes()), dtype="<i2")
        channels = w.getnchannels()
    audio = raw.reshape(-1, channels).mean(axis=1) if channels > 1 else raw
    return (audio / 32768.0).astype(np.float32)


def write_wav(path: Path, audio: np.ndarray, rate: int = RATE) -> None:
    pcm = (np.clip(audio, -1, 1) * 32767).astype("<i2")
    with wave.open(str(path), "wb") as w:
        w.setnchannels(1)
        w.setsampwidth(2)
        w.setframerate(rate)
        w.writeframes(pcm.tobytes())


def mix_at_snr(clean: np.ndarray, noise: np.ndarray, snr_db: float) -> np.ndarray:
    """Loop or trim noise to the clean length and scale it to the requested SNR."""
    reps = int(np.ceil(len(clean) / len(noise)))
    noise = np.tile(noise, reps)[: len(clean)]
    clean_power = np.mean(clean**2)
    noise_power = np.mean(noise**2) or 1e-12
    scale = np.sqrt(clean_power / (noise_power * 10 ** (snr_db / 10)))
    return (clean + noise * scale).astype(np.float32)
```

```python
"""Voice activity: Silero VAD v5 scores 32 ms frames; Segmenter turns scores into utterances."""

from collections import deque
from pathlib import Path

import numpy as np
import onnxruntime as ort

RATE = 16000
FRAME = 512
CONTEXT = 64  # Silero v5 expects the previous 64 samples in front of each frame


class SileroVad:
    def __init__(self, model_path: Path) -> None:
        opts = ort.SessionOptions()
        opts.intra_op_num_threads = 1
        opts.inter_op_num_threads = 1
        self.sess = ort.InferenceSession(
            str(model_path), sess_options=opts, providers=["CPUExecutionProvider"]
        )
        self.sr = np.array(RATE, dtype=np.int64)
        self.reset()

    def reset(self) -> None:
        self.state = np.zeros((2, 1, 128), dtype=np.float32)
        self.context = np.zeros((1, CONTEXT), dtype=np.float32)

    def prob(self, frame: np.ndarray) -> float:
        x = np.concatenate([self.context, frame.reshape(1, -1).astype(np.float32)], axis=1)
        out, self.state = self.sess.run(None, {"input": x, "state": self.state, "sr": self.sr})
        self.context = x[:, -CONTEXT:]
        return float(out[0][0])


class Segmenter:
    """Speech starts after `start_frames` voiced frames and ends after `end_silence_frames`
    quiet ones (with hysteresis), or at `max_frames`. Pre-roll keeps the first syllable."""

    def __init__(
        self,
        threshold: float = 0.5,
        start_frames: int = 3,
        end_silence_frames: int = 22,
        max_frames: int = 470,
        pre_roll: int = 8,
    ) -> None:
        self.threshold = threshold
        self.start_frames = start_frames
        self.end_silence_frames = end_silence_frames
        self.max_frames = max_frames
        self.pre_roll = pre_roll
        self.reset()

    def reset(self) -> None:
        self.active = False
        self._voiced = 0
        self._silent = 0
        self._buf: list[np.ndarray] = []
        self._pre: deque[np.ndarray] = deque(maxlen=self.pre_roll)

    def current_audio(self) -> np.ndarray:
        return np.concatenate(self._buf) if self._buf else np.zeros(0, np.float32)

    def push(self, frame: np.ndarray, prob: float) -> None | str | tuple[str, np.ndarray]:
        if not self.active:
            self._pre.append(frame)
            self._voiced = self._voiced + 1 if prob >= self.threshold else 0
            if self._voiced >= self.start_frames:
                self.active = True
                self._buf = list(self._pre)
                self._silent = 0
                return "start"
            return None
        self._buf.append(frame)
        quiet = prob < self.threshold - 0.15
        self._silent = self._silent + 1 if quiet else 0
        if self._silent >= self.end_silence_frames or len(self._buf) >= self.max_frames:
            audio = self.current_audio()
            self.reset()
            return ("end", audio)
        return None
```

```python
"""Speech to text on the device (no audio leaves the Pi)."""

from typing import Protocol

import numpy as np


class Transcriber(Protocol):
    def transcribe(self, audio: np.ndarray) -> str: ...


class MoonshineTranscriber:
    def __init__(self, model_name: str = "moonshine/base") -> None:
        from moonshine_onnx import MoonshineOnnxModel, load_tokenizer

        self.model = MoonshineOnnxModel(model_name=model_name)
        self.tokenizer = load_tokenizer()

    def transcribe(self, audio: np.ndarray) -> str:
        if len(audio) < 1600:  # under 0.1 s: nothing worth decoding
            return ""
        tokens = self.model.generate(audio[np.newaxis, :].astype(np.float32))
        return self.tokenizer.decode_batch(tokens)[0].strip()


def make_transcriber(voice_cfg: dict) -> Transcriber:
    return MoonshineTranscriber(voice_cfg["stt_model"])
```

```python
"""Speakerphone capture: 16 kHz mono float32 frames of 512 samples (32 ms)."""

import time
from collections.abc import Callable

import numpy as np
import sounddevice as sd

from voice.vad import FRAME, RATE


def find_input_device(name: str) -> int | None:
    """Index of the first input device whose name contains `name`; None means the default."""
    if not name:
        return None
    for index, dev in enumerate(sd.query_devices()):
        if dev["max_input_channels"] > 0 and name.lower() in dev["name"].lower():
            return index
    raise LookupError(f"No input device matching {name!r}")


class Microphone:
    def __init__(self, device_name: str) -> None:
        self.device_name = device_name
        self.on_frame: Callable[[np.ndarray], None] = lambda frame: None
        self.muted = True
        self.last_frame_at = 0.0
        self._stream: sd.InputStream | None = None

    def open(self) -> None:
        self.close()
        self._stream = sd.InputStream(
            device=find_input_device(self.device_name),
            channels=1,
            samplerate=RATE,
            blocksize=FRAME,
            dtype="float32",
            callback=self._callback,
        )
        self._stream.start()
        self.last_frame_at = time.monotonic()

    def close(self) -> None:
        if self._stream is not None:
            try:
                self._stream.stop()
                self._stream.close()
            finally:
                self._stream = None

    def _callback(self, indata, frames, time_info, status) -> None:
        self.last_frame_at = time.monotonic()
        if not self.muted:
            self.on_frame(indata[:, 0].copy())
```

```python
"""Mic -> VAD -> STT on a worker thread; results go to a VoiceSink on the asyncio loop."""

import asyncio
import logging
import queue
import threading
import time
from collections.abc import Callable
from typing import Protocol

import numpy as np

log = logging.getLogger("kiosk.voice")


class VoiceSink(Protocol):
    async def speech_started(self) -> None: ...
    async def partial(self, text: str) -> None: ...
    async def final(self, text: str) -> None: ...
    async def mic_status(self, ok: bool) -> None: ...


class VoicePipeline:
    def __init__(
        self,
        mic,
        vad,
        segmenter,
        transcriber,
        sink: VoiceSink,
        loop: asyncio.AbstractEventLoop,
        partial_every_s: float = 1.5,
        stall_s: float = 2.0,
        clock: Callable[[], float] = time.monotonic,
        check_every_s: float = 1.0,
        reopen_every_s: float = 5.0,
    ) -> None:
        self.mic = mic
        self.vad = vad
        self.seg = segmenter
        self.stt = transcriber
        self.sink = sink
        self.loop = loop
        self.partial_every_s = partial_every_s
        self.stall_s = stall_s
        self.clock = clock
        self.check_every_s = check_every_s
        self.reopen_every_s = reopen_every_s
        self._frames: queue.Queue[np.ndarray] = queue.Queue(maxsize=200)
        self._generation = 0  # bumped on every mute; stale frames are dropped
        self._running = threading.Event()
        self._threads: list[threading.Thread] = []
        self.mic.on_frame = self._on_frame

    # ---- control ----

    def start(self) -> None:
        self._running.set()
        for target, name in ((self._work, "voice-worker"), (self._watch, "voice-watch")):
            t = threading.Thread(target=target, name=name, daemon=True)
            t.start()
            self._threads.append(t)

    def stop(self) -> None:
        self._running.clear()
        for t in self._threads:
            t.join(timeout=2)
        self._threads.clear()
        self.mic.close()

    def set_muted(self, muted: bool) -> None:
        self.mic.muted = muted
        if muted:
            self._generation += 1
            while True:
                try:
                    self._frames.get_nowait()
                except queue.Empty:
                    break

    # ---- threads ----

    def _on_frame(self, frame: np.ndarray) -> None:
        if self.mic.muted:
            return
        try:
            self._frames.put_nowait((self._generation, frame))
        except queue.Full:
            log.warning("voice worker is behind; dropping audio")

    def _post(self, coro) -> None:
        asyncio.run_coroutine_threadsafe(coro, self.loop)

    def _work(self) -> None:
        generation = self._generation
        last_partial = 0.0
        while self._running.is_set():
            try:
                gen, frame = self._frames.get(timeout=0.2)
            except queue.Empty:
                continue
            if gen != self._generation or self.mic.muted:
                continue
            if gen != generation:
                generation = gen
                self.seg.reset()
                self.vad.reset()
            event = self.seg.push(frame, self.vad.prob(frame))
            now = self.clock()
            if event == "start":
                last_partial = now
                self._post(self.sink.speech_started())
            elif isinstance(event, tuple):
                text = self.stt.transcribe(event[1])
                if gen == self._generation:
                    self._post(self.sink.final(text))
            elif (
                self.seg.active
                and self.partial_every_s
                and now - last_partial >= self.partial_every_s
            ):
                last_partial = now
                text = self.stt.transcribe(self.seg.current_audio())
                if text and gen == self._generation:
                    self._post(self.sink.partial(text))

    def _watch(self) -> None:
        ok: bool | None = None
        last_open = -1e9
        while self._running.is_set():
            now = self.clock()
            healthy = now - self.mic.last_frame_at < self.stall_s
            if not healthy and now - last_open >= self.reopen_every_s:
                last_open = now
                try:
                    self.mic.open()
                    healthy = True
                except Exception as e:  # device unplugged or busy
                    log.warning("microphone unavailable: %s", e)
            if healthy != ok:
                ok = healthy
                self._post(self.sink.mic_status(ok))
            time.sleep(self.check_every_s)
```

The real `Microphone` starts with `last_frame_at = 0`, so the watcher opens it on its first pass; `start()` does not open it directly. A successful reopen counts as healthy straight away; if the device then delivers no frames, the next stall check reports failure again.

```python
"""Service mode: kiosk-voice runs separately and talks to kiosk-core over ZeroMQ."""

import asyncio
import logging

import zmq.asyncio

from core.bus import Publisher, Subscriber
from core.runtime import Runtime

log = logging.getLogger("kiosk.voice_link")


class VoiceLink:
    def __init__(
        self,
        runtime: Runtime,
        ctx: zmq.asyncio.Context,
        voice_endpoint: str,
        core_pub: Publisher,
    ) -> None:
        self.rt = runtime
        self.core_pub = core_pub
        self.sub = Subscriber(ctx, [voice_endpoint],
                              ["vad.start", "stt.partial", "stt.final", "mic.status"])
        self._task: asyncio.Task | None = None
        runtime.mic_listeners.append(self._on_mute)

    def start(self) -> None:
        self._task = asyncio.create_task(self._loop())

    async def stop(self) -> None:
        if self._task:
            self._task.cancel()
            await asyncio.gather(self._task, return_exceptions=True)
        self.sub.close()

    def _on_mute(self, muted: bool) -> None:
        asyncio.get_running_loop().create_task(self.core_pub.publish("mic.mute", {"muted": muted}))

    async def _loop(self) -> None:
        while True:
            topic, payload = await self.sub.recv()
            try:
                if topic == "vad.start":
                    await self.rt.speech_started()
                elif topic == "stt.partial":
                    await self.rt.partial(payload["text"])
                elif topic == "stt.final":
                    await self.rt.final(payload["text"])
                elif topic == "mic.status":
                    await self.rt.mic_status(bool(payload["ok"]))
            except Exception:
                log.exception("bad voice message %s", topic)
```

```python
"""kiosk-voice: mic, VAD and STT as a separate service (used when [voice] mode = "service")."""

import argparse
import asyncio
import logging
import time

import zmq.asyncio

from core.bus import Publisher, Subscriber
from core.config import load_config
from core.paths import models_dir
from voice.mic import Microphone
from voice.pipeline import VoicePipeline
from voice.stt import make_transcriber
from voice.vad import FRAME, RATE, Segmenter, SileroVad


class BusSink:
    def __init__(self, pub: Publisher) -> None:
        self.pub = pub

    async def speech_started(self) -> None:
        await self.pub.publish("vad.start", {"ts": time.time()})

    async def partial(self, text: str) -> None:
        await self.pub.publish("stt.partial", {"text": text})

    async def final(self, text: str) -> None:
        await self.pub.publish("vad.end", {"ts": time.time()})
        await self.pub.publish("stt.final", {"text": text})

    async def mic_status(self, ok: bool) -> None:
        await self.pub.publish("mic.status", {"ok": ok})


def build_pipeline(cfg: dict, sink, loop, mic_device: str | None = None) -> VoicePipeline:
    v = cfg["voice"]
    frame_ms = FRAME * 1000 / RATE
    return VoicePipeline(
        Microphone(v["mic_device"] if mic_device is None else mic_device),
        SileroVad(models_dir(cfg) / "silero_vad.onnx"),
        Segmenter(threshold=v["vad_threshold"],
                  end_silence_frames=round(v["end_silence_ms"] / frame_ms)),
        make_transcriber(v),
        sink,
        loop,
        partial_every_s=v["partial_every_s"],
    )


async def serve(cfg: dict) -> None:
    ctx = zmq.asyncio.Context()
    pub = Publisher(ctx, cfg["bus"]["voice_pub"])
    sub = Subscriber(ctx, [cfg["bus"]["core_pub"]], ["mic.mute"])
    pipeline = build_pipeline(cfg, BusSink(pub), asyncio.get_running_loop())
    pipeline.start()
    try:
        while True:
            _, payload = await sub.recv()
            pipeline.set_muted(bool(payload["muted"]))
    finally:
        pipeline.stop()


def main() -> None:
    parser = argparse.ArgumentParser(prog="python -m voice")
    parser.add_argument("--config", default="config.toml")
    args = parser.parse_args()
    logging.basicConfig(level=logging.INFO)
    asyncio.run(serve(load_config(args.config)))


if __name__ == "__main__":
    main()
```

```python
"""Benchmarks for the bench Pi (baseline 11.4). Prints one JSON object per result.

    uv run python -m tools.bench embed
    uv run python -m tools.bench stt tests/audio/what-courses.wav
    uv run python -m tools.bench vad
    uv run python -m tools.bench system --seconds 600 --every 5 > system.jsonl
"""

import argparse
import json
import os
import statistics
import time
from pathlib import Path

import numpy as np

from core.config import load_config
from core.paths import models_dir


def emit(**values) -> None:
    print(json.dumps(values), flush=True)


def bench_embed(cfg: dict) -> None:
    from core.rag.embed import FastEmbedder

    t0 = time.perf_counter()
    e = FastEmbedder(cfg["rag"]["embed_model"], cache_dir=models_dir(cfg) / "fastembed")
    emit(test="embed.load", seconds=round(time.perf_counter() - t0, 3))
    passages = [f"IIT offers programme number {i} in computing and business." * 8
                for i in range(50)]
    t0 = time.perf_counter()
    e.passages(passages)
    emit(test="embed.passages", n=50, seconds=round(time.perf_counter() - t0, 3))
    times = []
    for i in range(20):
        t0 = time.perf_counter()
        e.query(f"What programmes does IIT offer {i}?")
        times.append(time.perf_counter() - t0)
    emit(test="embed.query", median_ms=round(statistics.median(times) * 1000, 1))


def bench_stt(cfg: dict, wav: Path) -> None:
    from voice.stt import make_transcriber
    from voice.wav import read_wav

    audio = read_wav(wav)
    stt = make_transcriber(cfg["voice"])
    stt.transcribe(audio)  # warm up
    times, text = [], ""
    for _ in range(5):
        t0 = time.perf_counter()
        text = stt.transcribe(audio)
        times.append(time.perf_counter() - t0)
    emit(test="stt", model=cfg["voice"]["stt_model"], audio_s=round(len(audio) / 16000, 2),
         median_ms=round(statistics.median(times) * 1000), text=text)


def bench_vad(cfg: dict) -> None:
    from voice.vad import FRAME, SileroVad

    vad = SileroVad(models_dir(cfg) / "silero_vad.onnx")
    frame = np.zeros(FRAME, np.float32)
    t0 = time.perf_counter()
    for _ in range(500):
        vad.prob(frame)
    per = (time.perf_counter() - t0) / 500
    emit(test="vad", ms_per_frame=round(per * 1000, 3), realtime_factor=round(0.032 / per, 1))


def read_temp() -> float | None:
    path = Path("/sys/class/thermal/thermal_zone0/temp")
    return int(path.read_text()) / 1000 if path.exists() else None


def read_mem_mb() -> tuple[int, int] | None:
    path = Path("/proc/meminfo")
    if not path.exists():
        return None
    info = dict(line.split(":", 1) for line in path.read_text().splitlines())
    total = int(info["MemTotal"].split()[0]) // 1024
    avail = int(info["MemAvailable"].split()[0]) // 1024
    return total, avail


def bench_system(seconds: int, every: int) -> None:
    end = time.time() + seconds
    while time.time() < end:
        mem = read_mem_mb()
        emit(test="system", ts=round(time.time()), temp_c=read_temp(),
             load1=os.getloadavg()[0] if hasattr(os, "getloadavg") else None,
             mem_used_mb=(mem[0] - mem[1]) if mem else None)
        time.sleep(every)


def main() -> None:
    parser = argparse.ArgumentParser(prog="python -m tools.bench")
    parser.add_argument("--config", default="config.toml")
    sub = parser.add_subparsers(dest="cmd", required=True)
    sub.add_parser("embed")
    sub.add_parser("vad")
    stt = sub.add_parser("stt")
    stt.add_argument("wav", type=Path)
    system = sub.add_parser("system")
    system.add_argument("--seconds", type=int, default=600)
    system.add_argument("--every", type=int, default=5)
    args = parser.parse_args()
    cfg = load_config(args.config)
    if args.cmd == "embed":
        bench_embed(cfg)
    elif args.cmd == "stt":
        bench_stt(cfg, args.wav)
    elif args.cmd == "vad":
        bench_vad(cfg)
    else:
        bench_system(args.seconds, args.every)


if __name__ == "__main__":
    main()
```

```python
"""Replay recorded questions mixed with hall noise through the full answer pipeline.

    uv run python -m tools.replay record q01          # speak a question; saves tests/audio/q01.wav
    uv run python -m tools.replay run --noise tests/audio/hall-noise.wav --snr 5

Reports, per question: transcript, top source, whether it matched the expected source,
and time from end of speech to the first answer sentence (STT + search + LLM).
"""

import argparse
import asyncio
import time
import tomllib
from pathlib import Path

import numpy as np

from core.answer import AnswerError
from core.config import load_config
from core.paths import models_dir
from core.services import build_services
from voice.stt import make_transcriber
from voice.vad import FRAME, Segmenter, SileroVad
from voice.wav import mix_at_snr, read_wav, write_wav

AUDIO = Path("tests/audio")


def segment_offline(audio: np.ndarray, vad, segmenter: Segmenter) -> list[np.ndarray]:
    parts = []
    for start in range(0, len(audio) - FRAME + 1, FRAME):
        frame = audio[start : start + FRAME]
        event = segmenter.push(frame, vad.prob(frame))
        if isinstance(event, tuple):
            parts.append(event[1])
    return parts


def record(name: str, seconds: float = 6.0) -> None:
    import sounddevice as sd

    print(f"Recording {seconds:.0f} s... speak now")
    audio = sd.rec(int(seconds * 16000), samplerate=16000, channels=1, dtype="float32")
    sd.wait()
    write_wav(AUDIO / f"{name}.wav", audio[:, 0])
    print(f"saved {AUDIO / name}.wav")


async def first_sentence(answerer, question: str) -> tuple[str, float]:
    t0 = time.perf_counter()
    try:
        async for sentence in answerer.answer(question, (), False):
            return sentence, time.perf_counter() - t0
    except AnswerError as e:
        return f"<{e.kind}>", time.perf_counter() - t0
    return "<empty>", time.perf_counter() - t0


def run(cfg: dict, noise_path: Path | None, snr: float) -> int:
    spec = tomllib.loads((AUDIO / "questions.toml").read_text(encoding="utf-8"))
    services = build_services(cfg, with_voice=False)
    vad = SileroVad(models_dir(cfg) / "silero_vad.onnx")
    stt = make_transcriber(cfg["voice"])
    noise = read_wav(noise_path) if noise_path else None
    failures = 0
    for item in spec["question"]:
        clean = read_wav(AUDIO / item["file"])
        audio = mix_at_snr(clean, noise, snr) if noise is not None else clean
        parts = segment_offline(audio, vad, Segmenter())
        if not parts:
            print(f"MISS {item['file']}: no speech detected")
            failures += 1
            continue
        t0 = time.perf_counter()
        text = stt.transcribe(parts[0])
        stt_s = time.perf_counter() - t0
        hits = services.kb.search(text, 1)
        top = hits[0].title if hits else "-"
        sentence, llm_s = asyncio.run(first_sentence(services.answerer, text))
        ok = top == item["source"]
        failures += not ok
        total = stt_s + llm_s
        print(f"{'ok ' if ok else 'MISS'} {total:5.2f}s (stt {stt_s:.2f})  {item['file']}: "
              f"{text!r} -> {top} | {sentence}")
    print(f"\n{len(spec['question']) - failures}/{len(spec['question'])} matched")
    return 1 if failures else 0


def main() -> None:
    parser = argparse.ArgumentParser(prog="python -m tools.replay")
    parser.add_argument("--config", default="config.toml")
    sub = parser.add_subparsers(dest="cmd", required=True)
    rec = sub.add_parser("record")
    rec.add_argument("name")
    r = sub.add_parser("run")
    r.add_argument("--noise", type=Path)
    r.add_argument("--snr", type=float, default=5.0)
    args = parser.parse_args()
    if args.cmd == "record":
        record(args.name)
    else:
        raise SystemExit(run(load_config(args.config), args.noise, args.snr))


if __name__ == "__main__":
    main()
```
