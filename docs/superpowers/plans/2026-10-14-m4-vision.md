# M4: Camera Presence Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** With the Pi camera, a visitor walking up is greeted once, a passer-by is not, the bot's eyes follow the visitor, departure ends the encounter, and a dead camera switches the kiosk to "Tap to talk" within 5 s. No frame is ever stored.

**Architecture:** kiosk-vision is its own process (pinned to cores 2-3 in M6). It captures 640x480 frames with picamera2, halves them to 320x240, runs YOLO11n person detection through NCNN or OpenVINO (no PyTorch), tracks boxes with a small IoU tracker, and turns tracks into `presence` messages (count in zone, approach, gaze target). kiosk-core's `VisionLink` converts them into controller `Presence` events, watches the heartbeat, and sends `vision.rate` so detection drops to 2 fps while a visitor is engaged.

**Tech Stack:** picamera2 (system package on the Pi), numpy, ncnn, openvino, pyzmq; ultralytics only in an `export` dependency group used on the laptop or in CI.

**Spec:** `docs/superpowers/specs/2026-10-14-mvp-build-design.md`; baseline sections 4.2, 6, 6.1, 7.2, 9. Builds on the M1 to M3 plans.

## Changes to the baseline

- **No Ultralytics runtime on the Pi.** Baseline 6 says "Ultralytics YOLO11n exported to NCNN INT8, built-in ByteTrack". The Ultralytics package requires PyTorch, which conflicts with "No PyTorch on the Pi". So the exported model runs directly through the `ncnn` and `openvino` Python runtimes, and a small IoU tracker replaces ByteTrack. That is enough for the few people in front of a kiosk.
- **Precision.** Ultralytics exports NCNN at FP32/FP16, not INT8. INT8 is used for the OpenVINO export. The day-one benchmark (Task 8) compares both, as the baseline already plans.
- **Gaze mirroring.** `[vision] mirror = true` flips the x axis so the eyes look at the visitor rather than mirror them. Set it on site.

## Review Focus

1. **Camera unplugged while a visitor is mid-conversation.** The conversation must not end as a walk-away; the kiosk switches to camera-off behaviour. (Task 7: `test_heartbeat_loss_reports_camera_down_without_walkaway`)
2. **Two people, one approaching.** The eyes and `approaching` must follow the approaching person, not the larger bystander. (Task 7: `test_target_prefers_the_approaching_track`)
3. **Person at the frame edge.** Someone in the outer side bands is not "in front" and must not count as in zone. (Task 4: `test_people_outside_centre_band_are_not_in_zone`)
4. **Detector flicker.** A box missing for one frame keeps its track id, so approach history survives. (Task 3: `test_track_survives_one_missed_frame`)
5. **Garbage model output.** NaN or negative widths from a bad export must not crash the service. (Task 2: `test_decode_ignores_invalid_boxes`)

```python
"""YOLO11n person detection through NCNN or OpenVINO. Frames are never stored."""

from dataclasses import dataclass
from pathlib import Path
from typing import Protocol

import numpy as np

SIZE = 320
PAD = 114


@dataclass(frozen=True)
class Box:
    x1: float
    y1: float
    x2: float
    y2: float
    score: float


class Detector(Protocol):
    def detect(self, rgb: np.ndarray) -> list[Box]: ...


def letterbox(rgb: np.ndarray, size: int = SIZE) -> tuple[np.ndarray, float, int, int]:
    h, w = rgb.shape[:2]
    scale = min(size / w, size / h, 1.0)
    nh, nw = round(h * scale), round(w * scale)
    if scale != 1.0:
        ys = (np.arange(nh) / scale).astype(int)
        xs = (np.arange(nw) / scale).astype(int)
        rgb = rgb[ys][:, xs]
    pad_y, pad_x = (size - nh) // 2, (size - nw) // 2
    canvas = np.full((size, size, 3), PAD, np.uint8)
    canvas[pad_y : pad_y + nh, pad_x : pad_x + nw] = rgb
    chw = np.ascontiguousarray(canvas.transpose(2, 0, 1), dtype=np.float32) / 255.0
    return chw, scale, pad_x, pad_y


def nms(boxes: np.ndarray, scores: np.ndarray, iou: float) -> list[int]:
    order = list(np.argsort(-scores))
    keep: list[int] = []
    area = (boxes[:, 2] - boxes[:, 0]) * (boxes[:, 3] - boxes[:, 1])
    while order:
        i = order.pop(0)
        keep.append(int(i))
        rest = []
        for j in order:
            xx1 = max(boxes[i, 0], boxes[j, 0])
            yy1 = max(boxes[i, 1], boxes[j, 1])
            xx2 = min(boxes[i, 2], boxes[j, 2])
            yy2 = min(boxes[i, 3], boxes[j, 3])
            inter = max(0.0, xx2 - xx1) * max(0.0, yy2 - yy1)
            union = area[i] + area[j] - inter
            if union <= 0 or inter / union < iou:
                rest.append(j)
        order = rest
    return keep


def decode(
    output: np.ndarray,
    scale: float,
    pad_x: int,
    pad_y: int,
    conf: float = 0.4,
    iou: float = 0.5,
) -> list[Box]:
    out = output[0] if output.ndim == 3 else output
    cx, cy, w, h, person = out[0], out[1], out[2], out[3], out[4]
    valid = (
        np.isfinite(cx) & np.isfinite(cy) & np.isfinite(w) & np.isfinite(h)
        & (w > 0) & (h > 0) & (person >= conf)
    )
    if not valid.any():
        return []
    cx, cy, w, h, score = cx[valid], cy[valid], w[valid], h[valid], person[valid]
    boxes = np.stack(
        [
            (cx - w / 2 - pad_x) / scale,
            (cy - h / 2 - pad_y) / scale,
            (cx + w / 2 - pad_x) / scale,
            (cy + h / 2 - pad_y) / scale,
        ],
        axis=1,
    )
    return [
        Box(float(b[0]), float(b[1]), float(b[2]), float(b[3]), float(score[i]))
        for i, b in ((k, boxes[k]) for k in nms(boxes, score, iou))
    ]


class NcnnDetector:
    def __init__(self, model_dir: Path, threads: int = 2, conf: float = 0.4) -> None:
        import ncnn

        self._ncnn = ncnn
        self.conf = conf
        self.net = ncnn.Net()
        self.net.opt.use_vulkan_compute = False
        self.net.opt.num_threads = threads
        self.net.load_param(str(model_dir / "model.ncnn.param"))
        self.net.load_model(str(model_dir / "model.ncnn.bin"))

    def detect(self, rgb: np.ndarray) -> list[Box]:
        chw, scale, pad_x, pad_y = letterbox(rgb)
        ex = self.net.create_extractor()
        ex.input("in0", self._ncnn.Mat(chw))
        _, out = ex.extract("out0")
        return decode(np.array(out), scale, pad_x, pad_y, self.conf)


class OpenVinoDetector:
    def __init__(self, model_dir: Path, threads: int = 2, conf: float = 0.4) -> None:
        import openvino as ov

        core = ov.Core()
        self.conf = conf
        self.model = core.compile_model(
            str(model_dir / "yolo11n.xml"), "CPU", {"INFERENCE_NUM_THREADS": threads}
        )

    def detect(self, rgb: np.ndarray) -> list[Box]:
        chw, scale, pad_x, pad_y = letterbox(rgb)
        out = self.model(chw[None])[0]
        return decode(np.asarray(out), scale, pad_x, pad_y, self.conf)


def make_detector(vision_cfg: dict, models_dir: Path) -> Detector:
    backend = vision_cfg["backend"]
    args = (vision_cfg["threads"], vision_cfg["conf"])
    if backend == "ncnn":
        return NcnnDetector(models_dir / "yolo" / "ncnn", *args)
    if backend == "openvino":
        return OpenVinoDetector(models_dir / "yolo" / "openvino", *args)
    raise ValueError(f"Unknown vision backend {backend!r}; use ncnn or openvino.")
```

```python
"""Minimal IoU tracker: enough for the few people in front of a kiosk."""

from collections import deque
from dataclasses import dataclass, field

from vision.detect import Box


@dataclass
class Track:
    id: int
    box: Box
    last_seen: float
    heights: deque = field(default_factory=deque)


def iou(a: Box, b: Box) -> float:
    ix = max(0.0, min(a.x2, b.x2) - max(a.x1, b.x1))
    iy = max(0.0, min(a.y2, b.y2) - max(a.y1, b.y1))
    inter = ix * iy
    union = (a.x2 - a.x1) * (a.y2 - a.y1) + (b.x2 - b.x1) * (b.y2 - b.y1) - inter
    return inter / union if union > 0 else 0.0


class IouTracker:
    def __init__(self, iou_min: float = 0.3, max_age_s: float = 0.7, history_s: float = 3.0):
        self.iou_min = iou_min
        self.max_age_s = max_age_s
        self.history_s = history_s
        self.tracks: list[Track] = []
        self._next_id = 1

    def update(self, boxes: list[Box], now: float) -> list[Track]:
        self.tracks = [t for t in self.tracks if now - t.last_seen <= self.max_age_s]
        pairs = sorted(
            ((iou(t.box, b), ti, bi) for ti, t in enumerate(self.tracks)
             for bi, b in enumerate(boxes)),
            reverse=True,
        )
        used_t, used_b = set(), set()
        for score, ti, bi in pairs:
            if score < self.iou_min:
                break
            if ti in used_t or bi in used_b:
                continue
            used_t.add(ti)
            used_b.add(bi)
            self._observe(self.tracks[ti], boxes[bi], now)
        for bi, b in enumerate(boxes):
            if bi not in used_b:
                track = Track(self._next_id, b, now)
                self._next_id += 1
                self._observe(track, b, now)
                self.tracks.append(track)
        return [t for t in self.tracks if t.last_seen == now]

    def _observe(self, track: Track, box: Box, now: float) -> None:
        track.box = box
        track.last_seen = now
        track.heights.append((now, box.y2 - box.y1))
        while track.heights and now - track.heights[0][0] > self.history_s:
            track.heights.popleft()
```

```python
"""Turns tracks into the `presence` message (baseline 4.2 and 7.2)."""

from dataclasses import dataclass

from vision.track import Track


@dataclass(frozen=True)
class ZoneConfig:
    min_box_height_frac: float = 0.25
    approach_growth: float = 0.15
    approach_window_s: float = 1.5
    centre_band: float = 0.5
    mirror: bool = True

    @classmethod
    def from_config(cls, v: dict) -> "ZoneConfig":
        return cls(v["min_box_height_frac"], v["approach_growth"], v["approach_window_s"],
                   v["centre_band"], v["mirror"])


def _approaching(track: Track, now: float, zone: ZoneConfig) -> bool:
    window = [(t, h) for t, h in track.heights if now - t <= zone.approach_window_s]
    if len(window) < 3 or now - window[0][0] < zone.approach_window_s * 0.6:
        return False
    first, last = window[0][1], window[-1][1]
    return first > 0 and (last - first) / first >= zone.approach_growth


def presence(tracks: list[Track], frame_w: int, frame_h: int, zone: ZoneConfig, now: float) -> dict:
    out = []
    for t in tracks:
        b = t.box
        h = (b.y2 - b.y1) / frame_h
        cx = ((b.x1 + b.x2) / 2 / frame_w) * 2 - 1
        if h < zone.min_box_height_frac or abs(cx) > zone.centre_band:
            continue
        face_y = b.y1 + 0.2 * (b.y2 - b.y1)
        cy = (face_y / frame_h) * 2 - 1
        out.append(
            {
                "id": t.id,
                "cx": round(-cx if zone.mirror else cx, 3),
                "cy": round(cy, 3),
                "h": round(h, 3),
                "approaching": _approaching(t, now, zone),
            }
        )
    return {"ts": now, "count": len(out), "tracks": out}
```

import os
import socket


def notify(message: str) -> None:
    addr = os.environ.get("NOTIFY_SOCKET")
    if not addr or not hasattr(socket, "AF_UNIX"):
        return
    if addr.startswith("@"):
        addr = "\0" + addr[1:]
    with socket.socket(socket.AF_UNIX, socket.SOCK_DGRAM) as sock:
        sock.sendto(message.encode(), addr)

```python
"""kiosk-vision main loop: capture, detect, track, publish presence (the heartbeat)."""

import asyncio
import dataclasses
import logging
import time
from collections.abc import Callable

from core.sdnotify import notify
from vision.camera import Camera
from vision.detect import Detector
from vision.presence import ZoneConfig, presence
from vision.track import IouTracker

log = logging.getLogger("kiosk.vision")


class VisionService:
    def __init__(
        self,
        camera_factory: Callable[[], Camera],
        detector: Detector,
        tracker: IouTracker,
        zone: ZoneConfig,
        pub,
        fps: float = 8.0,
        preview_hz: float = 4.0,
        retry_s: float = 10.0,
        clock: Callable[[], float] = time.monotonic,
    ) -> None:
        self.camera_factory = camera_factory
        self.detector = detector
        self.tracker = tracker
        self.zone = zone
        self.pub = pub
        self.fps = fps
        self.preview_hz = preview_hz
        self.retry_s = retry_s
        self.clock = clock
        self._camera_ok: bool | None = None

    async def _set_camera(self, ok: bool) -> None:
        if ok != self._camera_ok:
            self._camera_ok = ok
            await self.pub.publish("camera", {"ok": ok})

    async def run(self) -> None:
        camera: Camera | None = None
        last_try = -1e9
        last_preview = -1e9
        while True:
            start = self.clock()
            notify("WATCHDOG=1")
            if camera is None:
                if start - last_try < self.retry_s:
                    await asyncio.sleep(min(0.5, self.retry_s))
                    continue
                last_try = start
                try:
                    camera = await asyncio.to_thread(self.camera_factory)
                    await self._set_camera(True)
                except Exception as e:
                    log.warning("camera unavailable: %s", e)
                    await self._set_camera(False)
                    continue
            try:
                frame = await asyncio.to_thread(camera.capture)
            except Exception as e:
                log.warning("camera failed: %s", e)
                await asyncio.to_thread(camera.close)
                camera = None
                last_try = start
                await self._set_camera(False)
                continue
            h, w = frame.shape[:2]
            boxes = await asyncio.to_thread(self.detector.detect, frame)
            del frame  # never kept beyond detection
            tracks = self.tracker.update(boxes, start)
            await self.pub.publish("presence", presence(tracks, w, h, self.zone, start))
            if start - last_preview >= 1 / self.preview_hz:
                last_preview = start
                await self.pub.publish(
                    "vision.preview",
                    {
                        "w": w,
                        "h": h,
                        "boxes": [[round(b.x1), round(b.y1), round(b.x2), round(b.y2),
                                   round(b.score, 2)] for b in boxes],
                        "zone": dataclasses.asdict(self.zone),
                    },
                )
            await asyncio.sleep(max(0.0, 1 / self.fps - (self.clock() - start)))

    async def control(self, sub) -> None:
        fields = {f.name for f in dataclasses.fields(ZoneConfig)}
        while True:
            topic, payload = await sub.recv()
            if topic == "vision.rate":
                self.fps = max(0.5, float(payload["fps"]))
            elif topic == "vision.config":
                changes = {k: v for k, v in payload.items() if k in fields}
                self.zone = dataclasses.replace(self.zone, **changes)
```

```python
"""kiosk-core side of kiosk-vision: presence in, frame rate and zone settings out."""

import asyncio
import logging
import time
from collections.abc import Callable

from core.controller.types import CameraStatus, Presence
from core.runtime import Runtime

log = logging.getLogger("kiosk.vision_link")


def to_presence(payload: dict) -> Presence:
    tracks = payload.get("tracks", [])
    if not tracks:
        return Presence(0)
    approaching = [t for t in tracks if t.get("approaching")]
    target = max(approaching or tracks, key=lambda t: t["h"])
    return Presence(len(tracks), bool(approaching), float(target["cx"]), float(target["cy"]))


class VisionLink:
    def __init__(
        self,
        runtime: Runtime,
        sub,
        core_pub,
        heartbeat_s: float = 5.0,
        clock: Callable[[], float] = time.monotonic,
    ) -> None:
        self.rt = runtime
        self.sub = sub
        self.core_pub = core_pub
        self.heartbeat_s = heartbeat_s
        self.clock = clock
        self.last_seen = clock()
        self.preview: dict | None = None
        self.fps = runtime.c.cfg.fps_idle
        self._tasks: list[asyncio.Task] = []
        runtime.vision_listeners.append(self._on_rate)

    def start(self) -> None:
        self._tasks = [asyncio.create_task(self._recv_loop()),
                       asyncio.create_task(self._watch_loop())]

    async def stop(self) -> None:
        for t in self._tasks:
            t.cancel()
        await asyncio.gather(*self._tasks, return_exceptions=True)

    def _on_rate(self, fps: float) -> None:
        self.fps = fps
        asyncio.get_running_loop().create_task(self.core_pub.publish("vision.rate", {"fps": fps}))

    async def publish_config(self, changes: dict) -> None:
        await self.core_pub.publish("vision.config", changes)

    async def on_message(self, topic: str, payload: dict) -> None:
        if topic == "presence":
            self.last_seen = self.clock()
            if not self.rt.c.camera_ok:
                await self.rt.handle(CameraStatus(True))
            await self.rt.handle(to_presence(payload))
        elif topic == "camera" and not payload.get("ok"):
            await self.rt.handle(CameraStatus(False))
        elif topic == "vision.preview":
            self.preview = payload

    async def check_heartbeat(self) -> None:
        if self.rt.c.camera_ok and self.clock() - self.last_seen > self.heartbeat_s:
            log.warning("no presence from kiosk-vision for %.0f s", self.heartbeat_s)
            await self.rt.handle(CameraStatus(False))

    async def _recv_loop(self) -> None:
        while True:
            topic, payload = await self.sub.recv()
            try:
                await self.on_message(topic, payload)
            except Exception:
                log.exception("bad vision message %s", topic)

    async def _watch_loop(self) -> None:
        while True:
            await asyncio.sleep(1.0)
            await self.check_heartbeat()
            # repeat the rate so a restarted kiosk-vision picks it up
            await self.core_pub.publish("vision.rate", {"fps": self.fps})
```

```python
"""Day-one detector benchmark on the Pi: fps, latency and freezes (baseline 11.4).

    uv run python -m tools.bench_vision --backend ncnn --minutes 30
"""

import argparse
import json
import statistics
import time

from core.config import load_config
from core.paths import models_dir
from tools.bench import read_temp
from vision.camera import PiCamera
from vision.detect import make_detector


def main() -> None:
    parser = argparse.ArgumentParser(prog="python -m tools.bench_vision")
    parser.add_argument("--config", default="config.toml")
    parser.add_argument("--backend", choices=["ncnn", "openvino"], required=True)
    parser.add_argument("--minutes", type=float, default=30)
    args = parser.parse_args()
    cfg = load_config(args.config)
    v = {**cfg["vision"], "backend": args.backend}
    detector = make_detector(v, models_dir(cfg))
    cam = PiCamera()
    times, gaps, max_temp = [], 0, 0.0
    end = time.monotonic() + args.minutes * 60
    last = time.monotonic()
    while time.monotonic() < end:
        frame = cam.capture()
        t0 = time.perf_counter()
        detector.detect(frame)
        times.append(time.perf_counter() - t0)
        now = time.monotonic()
        if now - last > 2.0:
            gaps += 1
        last = now
        max_temp = max(max_temp, read_temp() or 0.0)
    cam.close()
    print(json.dumps({
        "backend": args.backend,
        "frames": len(times),
        "median_ms": round(statistics.median(times) * 1000, 1),
        "p99_ms": round(sorted(times)[int(len(times) * 0.99)] * 1000, 1),
        "max_fps": round(1 / statistics.median(times), 1),
        "freezes_over_2s": gaps,
        "max_temp_c": max_temp,
    }))


if __name__ == "__main__":
    main()
```
